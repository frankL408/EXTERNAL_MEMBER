import mysql.connector

from datetime import datetime

# Configuration
MYSQL_CONFIG = {
    'host': 'localhost',
    'user': 'keystone',
    'password': 'Ryan1128',
    'database': 'keystone'
}

PROMETHEUS_URL = 'http://localhost:9090'

def check_mysql_connections():
    """Check Keystone database connectivity"""
    try:
        conn = mysql.connector.connect(**MYSQL_CONFIG)
        cursor = conn.cursor()
        cursor.execute("SHOW STATUS LIKE 'Threads_connected'")
        connections = cursor.fetchone()[1]
        conn.close()
        return int(connections)
    except Exception as e:
        print(f"MySQL Error: {str(e)}")
        return -1

def query_prometheus_metrics():
    """Query Prometheus for compliance data"""
    try:
        prom = PrometheusConnect(url=PROMETHEUS_URL)
        
        # Example: Get count of non-compliant ports
        non_compliant = prom.custom_query('count(iso27005_port_compliance{compliant="false"})')
        nc_count = int(non_compliant[0]['value'][1]) if non_compliant else 0
        
        # Example: Get NTP sync status
        ntp_offset = prom.custom_query('abs(iso27005_ntp_offset_seconds)')
        offset = float(ntp_offset[0]['value'][1]) if ntp_offset else 0.0
        
        return {
            'non_compliant_ports': nc_count,
            'ntp_offset_seconds': offset
        }
    except Exception as e:
        print(f"Prometheus Error: {str(e)}")
        return None

def generate_report():
    """Generate compliance report"""
    report = {
        'timestamp': datetime.now().isoformat(),
        'mysql_connections': check_mysql_connections(),
        **query_prometheus_metrics()
    }
    
    # Format as CSV (or modify for PDF/other formats)
    csv_output = f"""Compliance Report - {report['timestamp']}
MySQL Connections: {report['mysql_connections']}
Non-Compliant Ports: {report.get('non_compliant_ports', 'N/A')}
NTP Offset (seconds): {report.get('ntp_offset_seconds', 'N/A')}
"""
    return csv_output

# compliance_metrics.py
from prometheus_client import start_http_server, Gauge
import time
import subprocess
import socket

# Create metrics
COMPLIANCE_STATUS = Gauge('compliance_status', 
                         'ISO 27005 Compliance Status',
                         ['standard', 'component', 'instance'])

UNAUTHORIZED_PORTS = Gauge('unauthorized_ports_count',
                         'Count of unauthorized open ports',
                         ['instance'])

def scan_ports():
    """Scan for open ports on localhost"""
    open_ports = []
    # Common ports to check (you can customize this list)
    ports_to_scan = [22, 21, 80, 443, 5000, 80, 9103, 8080, 8774, 8775, 9292, 9696]
    
    for port in ports_to_scan:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(1)  # 1 second timeout
            result = s.connect_ex(('127.0.0.1', port))
            if result == 0:
                open_ports.append(port)
    return open_ports

def check_ports():
    """Check for unauthorized ports"""
    allowed_ports = {1883, 3000, 3306, 5001, 5432, 631, 6379, 9090, 9100, 9182, 9392, 9559}
    open_ports = scan_ports()  # Now this function exists
    unauthorized = [p for p in open_ports if p not in allowed_ports]
    
    UNAUTHORIZED_PORTS.labels(instance=socket.gethostname()).set(len(unauthorized))
    
    if len(unauthorized) > 0:
        COMPLIANCE_STATUS.labels(standard='ISO27005-2022',
                               component='port_protection',
                               instance=socket.gethostname()).set(0)
    else:
        COMPLIANCE_STATUS.labels(standard='ISO27005-2022',
                               component='port_protection',
                               instance=socket.gethostname()).set(1)

def check_ntp():
    """Check NTP synchronization"""
    try:
        result = subprocess.run(['ntpstat'], stdout=subprocess.PIPE)
        if result.returncode == 0:
            COMPLIANCE_STATUS.labels(standard='ISO27005-2022',
                                   component='clock_sync',
                                   instance=socket.gethostname()).set(1)
        else:
            COMPLIANCE_STATUS.labels(standard='ISO27005-2022',
                                   component='clock_sync',
                                   instance=socket.gethostname()).set(0)
    except:
        COMPLIANCE_STATUS.labels(standard='ISO27005-2022',
                               component='clock_sync',
                               instance=socket.gethostname()).set(0)

if __name__ == '__main__':
    start_http_server(8000)
    while True:
        check_ports()
        check_ntp()
        time.sleep(60)
